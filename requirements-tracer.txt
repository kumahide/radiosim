# RadioSim Tracer (apps/tracer) only — NOT part of the main app.
#
# Why a separate file:
#   requirements.txt is what goes INTO the packaged RadioSim exe. The Tracer is a
#   separate app that is never bundled (radiosim.spec must not reference apps/),
#   so its runtime needs stay out of the main app's dependency surface.
#   The Tracer's tests do not need anything here (the serial port is only opened
#   by `python -m apps.tracer.cli record`), so CI does not install this file.
#
# Install (into the project venv):
#   python -m pip install -r requirements-tracer.txt

# --- Serial port (COM) access for the RX board ---
pyserial==3.5
